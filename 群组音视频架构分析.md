```java
// 1. 发起群组通话
public void initiateGroupCall(String groupId, List<String> memberIds, boolean isVideo) {
    // 初始化成员列表
    groupMembers.clear();
    for (String memberId : memberIds) {
        GroupCallMember member = new GroupCallMember(memberId);
        member.setState(CallMemberState.INVITING);
        groupMembers.add(member);
    }
    
    // 创建并发送群组邀请信令
    MultiPartySignaling signaling = createGroupInviteSignaling(memberIds, isVideo);
    sendGroupSignaling(Constants.MsgType.multiPartyInvite, signaling);
    
    // 准备群组房间连接
    prepareGroupRoom(signaling);
}

// 2. 处理群组信令
public void handleGroupSignaling(MultiPartySignaling signaling) {
    // 使用去重组件处理，避免重复信令
    deduplicator.handleSignalingWithDeduplication(signaling, this::processGroupSignaling);
}

// 3. 信令类型处理分发
private void processGroupSignaling(MultiPartySignaling signaling) {
    switch (signaling.getType()) {
        case "multiPartyInvite": handleGroupInvite(signaling); break;
        case "multiPartyAccept": handleGroupAccept(signaling); break;
        case "multiPartyReject": handleGroupReject(signaling); break;
        case "multiPartyHangup": handleGroupHangup(signaling); break;
        case "multiPartyMemberJoin": handleMemberJoin(signaling); break;
        case "multiPartyMemberLeave": handleMemberLeave(signaling); break;
    }
}
```

#### GroupMemberAdapter (成员适配器)
```java
public class GroupMemberAdapter extends RecyclerView.Adapter<GroupMemberAdapter.MemberViewHolder> {
    private final List<GroupCallMember> memberList = new ArrayList<>();
    
    // Week 2 Day 6: 集成MultiStreamManager自动优先级管理
    @Override
    public void onBindViewHolder(MemberViewHolder holder, int position) {
        GroupCallMember member = memberList.get(position);
        holder.bind(member);
        
        // 注册视频流到MultiStreamManager
        if (member.isVideoVisible()) {
            StreamPriority priority = determineStreamPriority(member);
            callViewModel.registerVideoStream(member.getUserId(), holder.videoRenderer, priority);
        }
    }
    
    // 确定流优先级
    private StreamPriority determineStreamPriority(GroupCallMember member) {
        if (member.isSpeaking()) return StreamPriority.HIGH;    // 说话者高优先级
        if (member.getNetworkQuality() <= 2) return StreamPriority.LOW; // 网络差低优先级
        return StreamPriority.NORMAL;                           // 默认普通优先级
    }
}
```

```kotlin
    private fun scheduleStreamPriorities() {
        val sortedParticipants = participants.sortedWith(
```
